{{/* Shortcode: posts_category Muestra un recuadro compacto con los 5 posts más recientes de una categoría. Uso: {{< categoria "nombre-de-la-categoria" >}} Uso con etiqueta personalizada: {{< categoria "nombre-de-la-categoria" "Ver también" >}} */}} {{- $cat := .Get 0 -}} {{- $label := .Get 1 | default (printf "Publicaciones sobre %s" $cat) -}} {{- $catSlug := $cat | lower | replaceRE "\\s+" "-" -}} {{- $catPage := site.GetPage (printf "categories/%s" $catSlug) -}} {{ if $catPage }} {{/* Excluir la página actual para evitar recursión infinita */}} {{- $currentPage := .Page -}} {{- $filtered := where $catPage.Pages.ByDate.Reverse "RelPermalink" "ne" $currentPage.RelPermalink -}} {{- $pages := $filtered | first 5 -}}

{{ $label }}

{{ range $i, $page := $pages }} {{- $images := $page.Resources.ByType "image" -}} {{- $thumbnail := $images.GetMatch "*feature*" -}} {{ with $thumbnail }} {{ else }}
{{ end }} {{ $page.Title | markdownify }} {{ end }}
{{ else }} ⚠ categoria: no se encontró la categoría {{ $cat }} (slug: {{ $catSlug }}) {{ end }}
