{{/* Shortcode: posts_tag Muestra un recuadro compacto con los 5 posts más recientes de un tag. Uso: {{< posts_tag "nombre-del-tag" >}} Uso con etiqueta personalizada: {{< posts_tag "nombre-del-tag" "Ver también" >}} */}} {{- $tag := .Get 0 -}} {{- $label := .Get 1 | default (printf "Publicaciones sobre %s" $tag) -}} {{- $tagSlug := $tag | lower | replaceRE "\\s+" "-" -}} {{- $tagPage := site.GetPage (printf "tags/%s" $tagSlug) -}} {{ if $tagPage }} {{/* Excluir la página actual para evitar recursión infinita */}} {{- $currentPage := .Page -}} {{- $filtered := where $tagPage.Pages.ByDate.Reverse "RelPermalink" "ne" $currentPage.RelPermalink -}} {{- $pages := $filtered | first 5 -}}

{{ $label }}

{{ range $i, $page := $pages }} {{- $images := $page.Resources.ByType "image" -}} {{- $thumbnail := $images.GetMatch "*feature*" -}} {{ with $thumbnail }} {{ else }}
{{ end }} {{ $page.Title | markdownify }} {{ end }}
{{ else }} ⚠ etiqueta: no se encontró el tag {{ $tag }} (slug: {{ $tagSlug }}) {{ end }}
