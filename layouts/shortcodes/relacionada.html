{{/* Shortcode: enlace_post Muestra un cuadro con el thumbnail, título y excerpt de otro post. Uso: {{< enlace_post "blog/nombre-del-post" >}} Uso con título personalizado: {{< enlace_post "blog/nombre-del-post" "Ver también" >}} El path es relativo a la carpeta content/, sin barra inicial. El segundo parámetro (opcional) es el texto encima del recuadro; por defecto: "Publicaciones relacionadas". */}} {{- $path := .Get 0 -}} {{- $label := .Get 1 | default "Publicaciones relacionadas" -}} {{- $page := site.GetPage $path -}} {{ if $page }} {{- $images := $page.Resources.ByType "image" -}} {{- $thumbnail := $images.GetMatch "*feature*" -}} {{/* Usar solo excerpt o description para evitar recursión por .Summary */}} {{- $summary := $page.Params.excerpt | default $page.Description -}}

{{ $label }}

{{ with $thumbnail }} {{ end }}

{{ $page.Title | markdownify }}

{{ with $summary }}

{{ . | markdownify | plainify | truncate 200 "…" }}

{{ end }}
{{ else }} ⚠ enlace_post: no se encontró la página {{ $path }} {{ end }}
